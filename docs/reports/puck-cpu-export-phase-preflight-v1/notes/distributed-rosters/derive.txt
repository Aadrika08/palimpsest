def tokens(text):
    result=[];i=0;line=1
    while i<len(text):
        start=i;at=line
        if text[i].isspace():i+=1
        elif text.startswith('//',i):
            end=text.find('\n',i);i=len(text) if end<0 else end
        elif text.startswith('/*',i):i=text.index('*/',i+2)+2
        elif text.startswith('R"',i):
            opening=text.index('(',i+2);delimiter=text[i+2:opening]
            closing=text.index(')'+delimiter+'"',opening+1)
            result.append((text[opening+1:closing],at,'raw'))
            i=closing+len(delimiter)+2
        elif text[i] in ('"',"'"):
            quote=text[i];i+=1
            while text[i]!=quote:
                if text[i]=='\\':i+=1
                i+=1
            i+=1
            result.append((text[start+1:i-1],at,'string' if quote=='"' else 'char'))
        elif text[i].isalpha() or text[i]=='_':
            i+=1
            while i<len(text) and (text[i].isalnum() or text[i]=='_'):i+=1
            result.append((text[start:i],at,'name'))
        else:result.append((text[i],at,'punct'));i+=1
        line+=text[start:i].count('\n')
    return result


def close(ts,index,opening,closing):
    depth=1;index+=1
    while depth:
        if ts[index][0]==opening:depth+=1
        elif ts[index][0]==closing:depth-=1
        index+=1
    return index-1


def extract(path):
    raw=path.read_bytes();text=raw.decode();ts=tokens(text);stack=[];conditions={}
    for line,value in enumerate(text.splitlines(),1):
        directive=re.match(r'\s*#(ifdef|ifndef|if|elif|else|endif)\b(.*)',value)
        if directive:
            op,arg=directive.groups();arg=arg.strip()
            if op=='ifdef':stack.append('defined('+arg+')')
            elif op=='ifndef':stack.append('!defined('+arg+')')
            elif op=='if':stack.append(arg)
            elif op=='elif':stack[-1]='elif('+arg+')'
            elif op=='else':stack[-1]='!('+stack[-1]+')'
            else:stack.pop()
        conditions[line]=list(stack)
    rows=[];seen=set();constants=[]
    def add(name,line,kind):
        if (name,line,kind) not in seen:
            seen.add((name,line,kind));rows.append(dict(name=name,line=line,kind=kind,conditions=conditions[line]))
    for i,(word,line,kind) in enumerate(ts):
        if word in ('module','m') and [x[0] for x in ts[i+1:i+4]]==['.','def','(']:
            cursor=i+1
            while [x[0] for x in ts[cursor:cursor+3]]==['.','def','(']:
                if ts[cursor+3][2]=='string':add(ts[cursor+3][0],ts[cursor+3][1],'function')
                cursor=close(ts,cursor+2,'(',')')+1
        if word in ('module','m') and [x[0] for x in ts[i+1:i+4]]==['.','attr','('] and ts[i+4][2]=='string':
            constants.append(dict(name=ts[i+4][0],line=line,conditions=conditions[line],kind='data assignment; inspect RHS separately'))
        if (word.endswith('class_') or word.endswith('enum_')) and ts[i+1][0]=='<':
            end=close(ts,i+1,'<','>')+1
            if ts[end][2]=='name':end+=1
            if ts[end][0]=='(' and ts[end+1][0] in ('module','m') and ts[end+2][0]==',' and ts[end+3][2]=='string':
                add(ts[end+3][0],ts[end+3][1],'enum' if word.endswith('enum_') else 'class')
    grouped={}
    for row in rows:
        key=' && '.join(row['conditions']) or 'unconditional'
        grouped.setdefault(key,set()).add(row['name'])
    return dict(source=dict(path=str(path.relative_to(ROOT)),size_bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()),
        literal_direct_dispatch=rows,dispatch_by_condition={k:sorted(v) for k,v in grouped.items()},
        direct_data_assignments=constants)

    upstream_commit='0fabc3ba44823f257e70ce397d989c8de5e362c1',rows=result)
(OUT/'literal-rosters.json').write_text(json.dumps(output,sort_keys=True,indent=2)+'\n')
